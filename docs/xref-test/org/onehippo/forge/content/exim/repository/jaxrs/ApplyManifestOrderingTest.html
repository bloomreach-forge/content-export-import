37  /**
38   * Tests the package-private applyManifestOrdering in ContentEximImportService.
39   *
40   * Root cause of the ordering bug:
41   *   baseFolder.getName().getPath() returned the ZIP's on-disk path
42   *   (e.g. /tmp/foo.zip), not the ZIP-internal root "/".  Rank keys got the
43   *   disk path prepended; f.getName().getPath() returned only the ZIP-internal
44   *   path — zero matches — stable sort was a no-op — VFS HashMap order prevailed.
45   *
46   * Fix: use getRelativeName() which strips the base-folder prefix correctly
47   *   regardless of how the ZIP was opened (zip:/path vs zip:file:///path).
48   */
49  class ApplyManifestOrderingTest {
50  
51      @TempDir
52      Path tempDir;
53  
54      private final ContentEximImportService service = new ContentEximImportService();
55  
56      /**
57       * Manifest lists paths in reverse-alphabetical order (z → m → a).
58       * applyManifestOrdering must return FileObject[] in manifest order, not VFS
59       * hash order and not alphabetical order.
60       */
61      @Test
62      void sortsFilesByManifestOrderNotVfsHashOrder() throws Exception {
63          List<String> manifestOrder = List.of(
64                  "content/documents/knowledgebase/docs/getting-started/z-page/z-page.json",
65                  "content/documents/knowledgebase/docs/getting-started/m-page/m-page.json",
66                  "content/documents/knowledgebase/docs/getting-started/a-page/a-page.json"
67          );
68  
69          Path zipFile = buildZipWithManifest(tempDir.resolve("ordered.zip"), manifestOrder);
70  
71          FileSystemManager vfs = VFS.getManager();
72          FileObject baseFolder = vfs.resolveFile("zip:" + zipFile.toUri());
73          FileObject[] jsonFiles = jsonFiles(baseFolder);
74  
75          FileObject[] sorted = service.applyManifestOrdering(baseFolder, jsonFiles);
76  
77          List<String> sortedNames = Arrays.stream(sorted)
78                  .map(f -> f.getName().getBaseName())
79                  .toList();
80  
81          // Manifest order: z-page, m-page, a-page — NOT alphabetical (a, m, z)
82          assertThat(sortedNames).containsExactly("z-page.json", "m-page.json", "a-page.json");
83      }
84  
85      /** Without a manifest the input array is returned unchanged (no crash, no reorder). */
86      @Test
87      void returnsUnchangedWhenNoManifestPresent() throws Exception {
88          List<String> entries = List.of(
89                  "content/documents/docs/a/a.json",
90                  "content/documents/docs/b/b.json"
91          );
92          Path zipFile = buildZipNoManifest(tempDir.resolve("no-manifest.zip"), entries);
93  
94          FileSystemManager vfs = VFS.getManager();
95          FileObject baseFolder = vfs.resolveFile("zip:" + zipFile.toUri());
96          FileObject[] files = jsonFiles(baseFolder);
97  
98          FileObject[] result = service.applyManifestOrdering(baseFolder, files);
99          assertThat(result).hasSize(2);
100     }
101 
102     // ─── helpers ─────────────────────────────────────────────────────────────
103 
104     private FileObject[] jsonFiles(FileObject baseFolder) throws Exception {
105         FileObject[] all = baseFolder.findFiles(new AllFileSelector());
106         return Arrays.stream(all)
107                 .filter(f -> {
108                     try { return f.getType() == FileType.FILE
109                                  && f.getName().getBaseName().endsWith(".json"); }
110                     catch (Exception e) { return false; }
111                 })
112                 .toArray(FileObject[]::new);
113     }
114 
115     private Path buildZipWithManifest(Path zip, List<String> jsonPaths) throws Exception {
116         try (OutputStream fos = Files.newOutputStream(zip);
117              ZipOutputStream zos = new ZipOutputStream(fos)) {
118             StringBuilder manifest = new StringBuilder("# EXIM manifest\n");
119             for (String p : jsonPaths) manifest.append(p).append("\n");
120             writeEntry(zos, "EXIM-INF/manifest.txt",
121                     manifest.toString().getBytes(StandardCharsets.UTF_8));
122             for (String p : jsonPaths) writeEntry(zos, p, "{}".getBytes());
123         }
124         return zip;
125     }
126 
127     private Path buildZipNoManifest(Path zip, List<String> jsonPaths) throws Exception {
128         try (OutputStream fos = Files.newOutputStream(zip);
129              ZipOutputStream zos = new ZipOutputStream(fos)) {
130             for (String p : jsonPaths) writeEntry(zos, p, "{}".getBytes());
131         }
132         return zip;
133     }
134 
135     private void writeEntry(ZipOutputStream zos, String name, byte[] data) throws Exception {
136         zos.putNextEntry(new ZipEntry(name));
137         zos.write(data);
138         zos.closeEntry();
139     }
